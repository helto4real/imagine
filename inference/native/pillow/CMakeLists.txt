cmake_minimum_required(VERSION 3.24)
project(imagine_pillow LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(core "${PILLOW_SOURCE}/src/libImaging")
add_library(imagine_pillow SHARED bridge.c "${core}/Storage.c" "${core}/Resample.c" "${core}/Convert.c" "${core}/Palette.c" "${core}/Copy.c" "${core}/Mode.c" "${core}/ConvertYCbCr.c")
target_include_directories(imagine_pillow PRIVATE shim "${core}")
target_compile_options(imagine_pillow PRIVATE -O3 -fwrapv -fvisibility=hidden -ffunction-sections -fdata-sections)
target_link_options(imagine_pillow PRIVATE -Wl,--gc-sections -Wl,--no-undefined)
target_link_libraries(imagine_pillow PRIVATE m pthread)
